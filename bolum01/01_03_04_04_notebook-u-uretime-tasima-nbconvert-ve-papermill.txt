# ─── nbconvert: Notebook'u farkli formatlara donustur ───────────
pip install nbconvert

# HTML raporu olustur
jupyter nbconvert --to html analiz.ipynb
jupyter nbconvert --to html --no-input analiz.ipynb  # Kod gizle

# PDF raporu (LaTeX gerektirir)
jupyter nbconvert --to pdf analiz.ipynb

# Python betigi
jupyter nbconvert --to script analiz.ipynb
# cikti: analiz.py

# Notebook'u calistirip cikti kaydet
jupyter nbconvert --to notebook --execute analiz.ipynb \
       --output analiz_calistirilmis.ipynb

# Markdown (teknik blog icin)
jupyter nbconvert --to markdown analiz.ipynb

# ─── Papermill: Parametrik Notebook Calistirma ───────────────────
pip install papermill

# Notebook icinde parametre hucresi isaretle:
# tags: parameters  (JupyterLab'de hucre ayarlarinda)
# alpha = 0.01      # Bu deger disaridan degistirilebilir
# n_estimators = 100

# Farkli parametrelerle calistir
papermill model_egitim.ipynb cikti/model_0.01.ipynb \
        -p alpha 0.01 -p n_estimators 200

papermill model_egitim.ipynb cikti/model_0.1.ipynb \
        -p alpha 0.1 -p n_estimators 500

# Toplu calistirma — Python ile
import papermill as pm

parametreler = [
    {"alpha": 0.001, "n_est": 100},
    {"alpha": 0.01,  "n_est": 200},
    {"alpha": 0.1,   "n_est": 500},
]

for i, params in enumerate(parametreler):
    pm.execute_notebook(
        "model_egitim.ipynb",
        f"cikti/model_deney_{i}.ipynb",
        parameters=params,
        kernel_name="python3"
    )
    print(f"Deney {i} tamamlandi: {params}")
